function exportarResultado() {
    const data = document.getElementById('resultado-data').textContent;
    const blob = new Blob([data], { type: 'application/json' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');

    a.download = 'analisis_sentimientos.json';
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    window.URL.revokeObjectURL(url);
}

// Mostrar loading al enviar formulario
document.getElementById('uploadForm').addEventListener('submit', function() {
    const fileInput = this.querySelector('input[type="file"]');
    const file = fileInput.files[0];
    
    document.getElementById('submitBtn').disabled = true;
    document.getElementById('loadingIndicator').style.display = 'block';
    
    if (file) {
        const sizeMB = (file.size / (1024 * 1024)).toFixed(1);
        const isVideo = /\.(mp4|mov|avi|mkv|webm)$/i.test(file.name);
        
        let loadingText = `Procesando archivo de ${sizeMB} MB...`;
        if (isVideo) {
            loadingText += ' Primero se extraerá el audio del video.';
        }
        if (sizeMB > 25) {
            loadingText += ' El archivo se dividirá en partes para procesar.';
        }
        document.getElementById('loadingText').textContent = loadingText;
    }
});

// Copiar texto al portapapeles
function copiarTexto(elementId) {
    const texto = document.getElementById(elementId).innerText;
    navigator.clipboard.writeText(texto).then(() => {
        alert('Texto copiado al portapapeles');
    });
}

// Descargar texto como archivo
function descargarTexto(elementId, filename) {
    const texto = document.getElementById(elementId).innerText;
    const blob = new Blob([texto], { type: 'text/plain;charset=utf-8' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename || 'transcripcion.txt';
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    window.URL.revokeObjectURL(url);
}

<script>
function copiarTexto() {
    const texto = document.querySelector('pre').innerText;
    navigator.clipboard.writeText(texto).then(() => {
        alert('Texto copiado al portapapeles');
    });
}

function descargarTexto() {
    const texto = document.querySelector('pre').innerText;
    const blob = new Blob([texto], { type: 'text/plain;charset=utf-8' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'transcripcion.txt';
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    window.URL.revokeObjectURL(url);
}
</script>
